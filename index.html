const express = require('express'); const Anthropic = require('@anthropic-ai/sdk'); const app = express(); app.use(express.json()); const client = new Anthropic({ apiKey: process.env.ANTHROPIC_API_KEY }); const SYSTEM_PROMPT = `You are IdentityGuard, an expert identity protection agent exclusively serving DEVONN LYNCH (legal name: two N's, no middle name, no suffix). You help with: - Credit freezes at Equifax, Experian, TransUnion - Fraud alerts and dispute letters - FTC Identity Theft Reports at IdentityTheft.gov - IRS Form 14039 guidance - SSN compromise response - Monitoring name variants that may be used fraudulently Always address the user as "Devonn". Be precise, actionable, and include direct URLs when referencing government resources. Never ask for full SSN — only last 4 digits if necessary.`; app.post('/api/chat', async (req, res) => { try { const { messages } = req.body; if (!messages || !Array.isArray(messages)) { return res.status(400).json({ error: 'messages array required' }); } const response = await client.messages.create({ model: 'claude-sonnet-4-20250514', max_tokens: 1024, system: SYSTEM_PROMPT, messages: messages }); const reply = response.content .filter(b => b.type === 'text') .map(b => b.text) .join('\n'); res.json({ reply }); } catch (err) { console.error('Anthropic API error:', err); res.status(500).json({ error: err.message || 'Internal server error' }); } }); app.listen(3000, () => console.log('IdentityGuard API running on :3000'));
